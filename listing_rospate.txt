type benchEnv struct {
	g          *crypto.GOSTProvider
	engine     *engine.SmartContractEngine
	ledger     *ledger.BadgerLedger
	issuerPriv *gost3410.PrivateKey
	oraclePriv *gost3410.PrivateKey
}
func setupBenchEnv(b *testing.B) *benchEnv {
	b.Helper()
	g := crypto.NewGOSTProvider()
	issuerPriv, issuerPub, err := g.GenerateKeyPair()
	if err != nil {
		b.Fatalf("issuer keypair: %v", err)
	}
	oraclePriv, oraclePub, err := g.GenerateKeyPair()
	if err != nil {
		b.Fatalf("oracle keypair: %v", err)
	}
	dir := filepath.Join(b.TempDir(), "bench-db")
	bl, err := ledger.NewBadgerLedger(dir)
	if err != nil {
		b.Fatalf("ledger: %v", err)
	}
	b.Cleanup(func() { _ = bl.Close() })
	eng, err := engine.NewSmartContractEngine(bl, g, issuerPub, oraclePub)
	if err != nil {
		b.Fatalf("engine: %v", err)
	}
	return &benchEnv{
		g:          g,
		engine:     eng,
		ledger:     bl,
		issuerPriv: issuerPriv,
		oraclePriv: oraclePriv,
	}
}
func runOneCycle(env *benchEnv, dealID, milHash, inn, kadastr string) error {
	token, err := env.engine.IssueColoredRubles(
		485_000_000, milHash, inn, kadastr, env.issuerPriv,
	)
	if err != nil {
		return fmt.Errorf("issue: %w", err)
	}
	if err := env.engine.LockInEscrow(
		dealID, token, engine.DefaultLockDuration,
	); err != nil {
		return fmt.Errorf("lock: %w", err)
	}
	digest := env.g.Streebog256([]byte(dealID + ":" + kadastr + ":true"))
	oracleSig, err := env.g.SignGOST3410(env.oraclePriv, digest)
	if err != nil {
		return fmt.Errorf("sign: %w", err)
	}
	if err := env.engine.ProcessRosreestrCallback(
		dealID, kadastr, true, oracleSig,
	); err != nil {
		return fmt.Errorf("callback: %w", err)
	}
	return nil
}
func BenchmarkFullCycle(b *testing.B) {
	env := setupBenchEnv(b)
	latencies := make([]time.Duration, 0, b.N)
	b.ResetTimer()
	start := time.Now()
	for i := 0; i < b.N; i++ {
		opStart := time.Now()
		dealID := fmt.Sprintf("DEAL-BENCH-%d", i)
		milHash := fmt.Sprintf("mil-%d", i)
		kadastr := fmt.Sprintf("77:01:%07d:1001", i%1_000_000)
		if err := runOneCycle(env, dealID, milHash, "7701982341", kadastr); err != nil {
			b.Fatalf("cycle[%d]: %v", i, err)
		}
		latencies = append(latencies, time.Since(opStart))
	}
	elapsed := time.Since(start)
	b.StopTimer()
	if b.N < 10 {
		return 
	}
	report(b, "SEQUENTIAL", len(latencies), elapsed, latencies)
}
func BenchmarkParallelFullCycle(b *testing.B) {
	env := setupBenchEnv(b)
	const numWorkers = 8
	opsPerWorker := b.N / numWorkers
	if opsPerWorker == 0 {
		opsPerWorker = 1
	}
	latencies := make([]time.Duration, b.N)
	var latIdx atomic.Uint64
	var successCount atomic.Uint64
	var wg sync.WaitGroup
	b.ResetTimer()
	start := time.Now()
	for w := 0; w < numWorkers; w++ {
		wg.Add(1)
		go func(workerID int) {
			defer wg.Done()
			for i := 0; i < opsPerWorker; i++ {
				opStart := time.Now()
				dealID := fmt.Sprintf("DEAL-PAR-%d-%d", workerID, i)
				milHash := fmt.Sprintf("mil-%d-%d", workerID, i)
				kadastr := fmt.Sprintf("77:01:%07d:1001", (workerID*1000+i)%1_000_000)
				if err := runOneCycle(env, dealID, milHash, "7701982341", kadastr); err != nil {
					continue
				}
				successCount.Add(1)
				idx := latIdx.Add(1) - 1
				if int(idx) < len(latencies) {
					latencies[idx] = time.Since(opStart)
				}
			}
		}(w)
	}
	wg.Wait()
	elapsed := time.Since(start)
	b.StopTimer()
	done := int(successCount.Load())
	if done == 0 {
		b.Fatal("no successful cycles")
	}
	if done < len(latencies) {
		latencies = latencies[:done]
	}
	if b.N < 10 {
		return
	}
	report(b, "PARALLEL (8 workers)", done, elapsed, latencies)
}
func report(
	b *testing.B,
	mode string,
	count int,
	elapsed time.Duration,
	latencies []time.Duration,
) {
	b.Helper()
	if len(latencies) == 0 {
		b.Fatal("no latencies collected")
	}
	tps := float64(count) / elapsed.Seconds()
	b.ReportMetric(tps, "TPS")
	sorted := make([]time.Duration, len(latencies))
	copy(sorted, latencies)
	sort.Slice(sorted, func(i, j int) bool { return sorted[i] < sorted[j] })
	p := func(q float64) time.Duration {
		if len(sorted) == 0 {
			return 0
		}
		idx := int(float64(len(sorted)-1) * q)
		return sorted[idx]
	}
	var total time.Duration
	for _, d := range sorted {
		total += d
	}
	avg := total / time.Duration(len(sorted))
	printReport(mode, count, elapsed, tps, avg, p(0.50), p(0.95), p(0.99), sorted[len(sorted)-1])
}
func printReport(
	mode string,
	count int,
	elapsed time.Duration,
	tps float64,
	avg, p50, p95, p99, mx time.Duration,
) {
	lines := []string{
		"",
		"+----------------------------------------------------------+",
		fmt.Sprintf("|  BENCHMARK RESULTS: DIGITAL NIS [%s]", padRight(mode, 20)),
		"+----------------------------------------------------------+",
		fmt.Sprintf("|  Full deal cycles      : %10d                          |", count),
		fmt.Sprintf("|  Total time            : %-14v                  |", elapsed.Round(time.Millisecond)),
		fmt.Sprintf("|  Throughput            : %10.2f TPS                     |", tps),
		"+----------------------------------------------------------+",
		"|  Latency (per full cycle)                                |",
		fmt.Sprintf("|    average : %-14v                              |", avg.Round(time.Microsecond)),
		fmt.Sprintf("|    p50     : %-14v                              |", p50.Round(time.Microsecond)),
		fmt.Sprintf("|    p95     : %-14v                              |", p95.Round(time.Microsecond)),
		fmt.Sprintf("|    p99     : %-14v                              |", p99.Round(time.Microsecond)),
		fmt.Sprintf("|    max     : %-14v                              |", mx.Round(time.Microsecond)),
		"+----------------------------------------------------------+",
		"|  Crypto    : GOST R 34.11-2012 / 34.10-2012              |",
		"|  Storage   : BadgerDB (embedded, LSM-tree)               |",
		"+----------------------------------------------------------+",
		"",
	}
	for _, l := range lines {
		fmt.Fprintln(os.Stdout, l)
	}
	if path := os.Getenv("NIS_BENCH_OUT"); path != "" {
		content := strings.Join(lines, "\n") + "\n"
		if err := os.WriteFile(path, []byte(content), 0o644); err != nil {
			fmt.Fprintf(os.Stderr, "benchmark: write %s: %v\n", path, err)
		} else {
			fmt.Fprintf(os.Stderr, "benchmark: results written to %s\n", path)
		}
	}
}
func padRight(s string, w int) string {
	for len(s) < w {
		s += " "
	}
	return s
}

package main
import (
	"encoding/hex"
	"errors"
	"flag"
	"fmt"
	"log"
	"time"
	badger "github.com/dgraph-io/badger/v4"
	"nis-core/internal/audit"
	"nis-core/internal/crypto"
	"nis-core/internal/engine"
	"nis-core/internal/ledger"
)
type Stats struct {
	Total    int
	Created  int
	Locked   int
	Released int
	Refunded int
}
func main() {
	dbPath := flag.String("db", "./data/nis-state", "path to BadgerDB")
	cmd := flag.String("cmd", "full-audit", "command: full-audit | prove")
	dealID := flag.String("deal", "", "dealID for prove command")
	flag.Parse()
	if err := run(*dbPath, *cmd, *dealID); err != nil {
		log.Fatalf("[auditor] %v", err)
	}
}
func run(dbPath, cmd, dealID string) error {
	g := crypto.NewGOSTProvider()
	opts := badger.DefaultOptions(dbPath).
		WithLogger(nil).
		WithLoggingLevel(badger.ERROR).
		WithValueLogFileSize(64 << 20)
	db, err := badger.Open(opts)
	if err != nil {
		return fmt.Errorf("open db: %w", err)
	}
	defer db.Close()
	switch cmd {
	case "full-audit":
		return fullAudit(g, db)
	case "prove":
		if dealID == "" {
			return errors.New("prove: -deal <dealID> is required")
		}
		return prove(g, db, dealID)
	default:
		return fmt.Errorf("unknown command %q (use full-audit or prove)", cmd)
	}
}
func loadAll(db *badger.DB) (map[string]*engine.EscrowContract, error) {
	contracts := make(map[string]*engine.EscrowContract)
	err := db.View(func(txn *badger.Txn) error {
		it := txn.NewIterator(badger.DefaultIteratorOptions)
		defer it.Close()
		prefix := []byte(ledger.EscrowKeyPrefix)
		for it.Seek(prefix); it.ValidForPrefix(prefix); it.Next() {
			item := it.Item()
			key := string(item.Key())
			dealID := key[len(ledger.EscrowKeyPrefix):]
			var payload []byte
			if err := item.Value(func(v []byte) error {
				payload = append([]byte(nil), v...)
				return nil
			}); err != nil {
				return err
			}
			c, err := engine.DeserializeEscrow(payload)
			if err != nil {
				return fmt.Errorf("deserialize %s: %w", dealID, err)
			}
			contracts[dealID] = c
		}
		return nil
	})
	if err != nil {
		return nil, err
	}
	return contracts, nil
}
func fullAudit(g *crypto.GOSTProvider, db *badger.DB) error {
	contracts, err := loadAll(db)
	if err != nil {
		return err
	}
	if len(contracts) == 0 {
		fmt.Println("Реестр пуст — нет ни одного эскроу-контракта.")
		return nil
	}
	tree, err := audit.BuildTree(g, contracts)
	if err != nil {
		return fmt.Errorf("build merkle: %w", err)
	}
	stats := Stats{Total: len(contracts)}
	for _, c := range contracts {
		switch c.Status {
		case engine.StatusCreated:
			stats.Created++
		case engine.StatusLocked:
			stats.Locked++
		case engine.StatusReleased:
			stats.Released++
		case engine.StatusRefunded:
			stats.Refunded++
		}
	}
	fmt.Println("========================================================")
	fmt.Println("  АУДИТ РЕЕСТРА «ЦИФРОВАЯ НИС»")
	fmt.Println("========================================================")
	fmt.Printf("  Время аудита       : %s\n",
		time.Now().UTC().Format(time.RFC3339))
	fmt.Printf("  Всего контрактов   : %d\n", stats.Total)
	fmt.Printf("    CREATED          : %d\n", stats.Created)
	fmt.Printf("    LOCKED           : %d\n", stats.Locked)
	fmt.Printf("    RELEASED         : %d\n", stats.Released)
	fmt.Printf("    REFUNDED         : %d\n", stats.Refunded)
	fmt.Println("--------------------------------------------------------")
	fmt.Printf("  Корневой хэш       : %s\n", hex.EncodeToString(tree.Tree.RootHash()))
	fmt.Printf("  Алгоритм           : ГОСТ Р 34.11-2012 (Стрибог-256)\n")
	fmt.Printf("  Листьев в дереве   : %d\n", tree.Tree.Size())
	fmt.Println("========================================================")
	return nil
}
func prove(g *crypto.GOSTProvider, db *badger.DB, dealID string) error {
	contracts, err := loadAll(db)
	if err != nil {
		return err
	}
	if len(contracts) == 0 {
		return errors.New("реестр пуст")
	}
	tree, err := audit.BuildTree(g, contracts)
	if err != nil {
		return fmt.Errorf("build merkle: %w", err)
	}
	idx := tree.IndexOf(dealID)
	if idx < 0 {
		return fmt.Errorf("сделка %q не найдена в реестре", dealID)
	}
	proof, err := tree.Tree.Prove(idx)
	if err != nil {
		return fmt.Errorf("prove: %w", err)
	}
	ok := crypto.VerifyProof(g, tree.Tree.RootHash(), proof)
	rec := contracts[dealID]
	fmt.Println("========================================================")
	fmt.Printf("  ДОКАЗАТЕЛЬСТВО ВКЛЮЧЕНИЯ СДЕЛКИ %s\n", dealID)
	fmt.Println("========================================================")
	fmt.Printf("  Лист (leaf hash)   : %s\n", hex.EncodeToString(proof.LeafHash))
	fmt.Printf("  Глубина дерева     : %d\n", len(proof.Siblings))
	fmt.Println("  Путь (sibling hashes):")
	for i, s := range proof.Siblings {
		side := "справа"
		if s.IsLeft {
			side = "слева "
		}
		fmt.Printf("    %2d) [%s] %s\n", i+1, side, hex.EncodeToString(s.Hash))
	}
	fmt.Println("--------------------------------------------------------")
	fmt.Printf("  Корневой хэш       : %s\n", hex.EncodeToString(tree.Tree.RootHash()))
	fmt.Printf("  Верификация        : %v\n", verdict(ok))
	fmt.Println("--------------------------------------------------------")
	fmt.Printf("  Сделка\n")
	fmt.Printf("    Статус           : %s\n", rec.Status)
	fmt.Printf("    Сумма (коп.)     : %d\n", rec.Token.Amount)
	fmt.Printf("    Токен (prefix)   : %s...\n", prefix(rec.Token.TokenID, 16))
	fmt.Printf("    Кадастр          : %s\n", rec.Token.CadastralNumber)
	fmt.Printf("    Создана          : %s\n",
		time.Unix(rec.CreatedAt, 0).UTC().Format(time.RFC3339))
	fmt.Printf("    Обновлена        : %s\n",
		time.Unix(rec.UpdatedAt, 0).UTC().Format(time.RFC3339))
	fmt.Println("========================================================")
	if !ok {
		return errors.New("верификация доказательства не прошла")
	}
	return nil
}
func verdict(ok bool) string {
	if ok {
		return "ПОДТВЕРЖДЕНО (proof valid)"
	}
	return "ОТКЛОНЕНО (proof invalid)"
}
func prefix(s string, n int) string {
	if len(s) <= n {
		return s
	}
	return s[:n]
}

package main
import (

type handlers struct {

}
func writeJSON(w http.ResponseWriter, status int, v any) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)
	if err := json.NewEncoder(w).Encode(v); err != nil {
		fmt.Fprintf(w, `{"error":"encode failed"}`)
	}
}
func writeErr(w http.ResponseWriter, status int, err error) {
	writeJSON(w, status, map[string]string{"error": err.Error()})
}
func mapEngineErr(err error) int {
	switch {
	case errors.Is(err, engine.ErrNotFound):
		return http.StatusNotFound
	case errors.Is(err, engine.ErrConflict):
		return http.StatusConflict
	case errors.Is(err, engine.ErrInvalidArgument):
		return http.StatusBadRequest
	case errors.Is(err, engine.ErrInvalidState):
		return http.StatusConflict
	case errors.Is(err, engine.ErrTokenMisuse),
		errors.Is(err, engine.ErrIssuerSigInvalid),
		errors.Is(err, engine.ErrOracleSigInvalid):
		return http.StatusForbidden
	case errors.Is(err, engine.ErrContractExpired):
		return http.StatusGone
	default:
		return http.StatusInternalServerError
	}
}
func (h *handlers) healthz(w http.ResponseWriter, r *http.Request) {
	writeJSON(w, http.StatusOK, map[string]any{
		"status":    "ok",
		"node_id":   h.app.Config.NodeID,
		"demo_mode": h.app.Config.DemoMode,
		"time":      time.Now().UTC().Format(time.RFC3339),
	})
}
func (h *handlers) getEscrow(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodGet {
		writeErr(w, http.StatusMethodNotAllowed, errors.New("method not allowed"))
		return
	}
	dealID := strings.TrimPrefix(r.URL.Path, "/api/v1/escrow/")
	if dealID == "" {
		writeErr(w, http.StatusBadRequest, errors.New("dealID is required"))
		return
	}
	c, err := h.app.Engine.GetEscrow(dealID)
	if err != nil {
		writeErr(w, mapEngineErr(err), err)
		return
	}
	writeJSON(w, http.StatusOK, c)
}
type issueRequest struct {
	Amount       uint64 `json:"amount"`
	MilitaryHash string `json:"military_hash"`
	DeveloperINN string `json:"developer_inn"`
	Kadastr      string `json:"kadastr"`
}
func (h *handlers) issue(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodPost {
		writeErr(w, http.StatusMethodNotAllowed, errors.New("method not allowed"))
		return
	}
	if !h.app.Config.DemoMode || h.app.IssuerPriv == nil {
		writeErr(w, http.StatusForbidden,
			errors.New("issue endpoint requires demo_mode with issuer key"))
		return
	}
	var req issueRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		writeErr(w, http.StatusBadRequest, fmt.Errorf("invalid JSON: %w", err))
		return
	}
	token, err := h.app.Engine.IssueColoredRubles(
		req.Amount, req.MilitaryHash, req.DeveloperINN, req.Kadastr,
		h.app.IssuerPriv,
	)
	if err != nil {
		writeErr(w, mapEngineErr(err), err)
		return
	}
	writeJSON(w, http.StatusCreated, token)
}
type lockRequest struct {
	DealID      string                  `json:"deal_id"`
	Token       engine.DigitalNISRubles `json:"token"`
	LockSeconds int64                   `json:"lock_seconds"`
}
func (h *handlers) lock(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodPost {
		writeErr(w, http.StatusMethodNotAllowed, errors.New("method not allowed"))
		return
	}
	var req lockRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		writeErr(w, http.StatusBadRequest, fmt.Errorf("invalid JSON: %w", err))
		return
	}
	dur := time.Duration(req.LockSeconds) * time.Second
	if dur <= 0 {
		dur = engine.DefaultLockDuration
	}
	if err := h.app.Engine.LockInEscrow(req.DealID, &req.Token, dur); err != nil {
		writeErr(w, mapEngineErr(err), err)
		return
	}
	c, _ := h.app.Engine.GetEscrow(req.DealID)
	writeJSON(w, http.StatusOK, c)
}
type callbackRequest struct {
	DealID    string `json:"deal_id"`
	Kadastr   string `json:"kadastr"`
	Success   bool   `json:"success"`
	Signature []byte `json:"signature"` 
}
func (h *handlers) rosreestrCallback(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodPost {
		writeErr(w, http.StatusMethodNotAllowed, errors.New("method not allowed"))
		return
	}
	var req callbackRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		writeErr(w, http.StatusBadRequest, fmt.Errorf("invalid JSON: %w", err))
		return
	}
	err := h.app.Engine.ProcessRosreestrCallback(
		req.DealID, req.Kadastr, req.Success, req.Signature,
	)
	if err != nil && !errors.Is(err, engine.ErrContractExpired) {
		writeErr(w, mapEngineErr(err), err)
		return
	}
	c, gerr := h.app.Engine.GetEscrow(req.DealID)
	if gerr != nil {
		writeErr(w, mapEngineErr(gerr), gerr)
		return
	}
	resp := map[string]any{"contract": c}
	if errors.Is(err, engine.ErrContractExpired) {
		resp["expired"] = true
	}
	writeJSON(w, http.StatusOK, resp)
}
type demoRunRequest struct {
	DealID       string `json:"deal_id"`
	Amount       uint64 `json:"amount"`
	MilitaryHash string `json:"military_hash"`
	DeveloperINN string `json:"developer_inn"`
	Kadastr      string `json:"kadastr"`
	LockSeconds  int64  `json:"lock_seconds,omitempty"`
	Reject       bool   `json:"reject,omitempty"`
}
type demoRunResponse struct {
	DealID          string                  `json:"deal_id"`
	Token           engine.DigitalNISRubles `json:"token"`
	OracleSignature string                  `json:"oracle_signature"` 
	ContractFinal   *engine.EscrowContract  `json:"contract_final"`
	Expired         bool                    `json:"expired,omitempty"`
}
func (h *handlers) demoRun(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodPost {
		writeErr(w, http.StatusMethodNotAllowed, errors.New("method not allowed"))
		return
	}
	if !h.app.Config.DemoMode {
		writeErr(w, http.StatusForbidden,
			errors.New("demo endpoint requires demo_mode: true"))
		return
	}
	if h.app.IssuerPriv == nil {
		writeErr(w, http.StatusForbidden,
			errors.New("demo endpoint requires issuer private key"))
		return
	}
	if h.app.OraclePriv == nil {
		writeErr(w, http.StatusForbidden,
			errors.New("demo endpoint requires oracle private key"))
		return
	}
	var req demoRunRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		writeErr(w, http.StatusBadRequest, fmt.Errorf("invalid JSON: %w", err))
		return
	}
	if req.DealID == "" || req.Amount == 0 ||
		req.MilitaryHash == "" || req.DeveloperINN == "" || req.Kadastr == "" {
		writeErr(w, http.StatusBadRequest,
			errors.New("deal_id, amount, military_hash, developer_inn, kadastr required"))
		return
	}
	token, err := h.app.Engine.IssueColoredRubles(
		req.Amount, req.MilitaryHash, req.DeveloperINN, req.Kadastr,
		h.app.IssuerPriv,
	)
	if err != nil {
		writeErr(w, mapEngineErr(err), err)
		return
	}
	lockDur := time.Duration(req.LockSeconds) * time.Second
	if lockDur <= 0 {
		lockDur = engine.DefaultLockDuration
	}
	if err := h.app.Engine.LockInEscrow(req.DealID, token, lockDur); err != nil {
		writeErr(w, mapEngineErr(err), err)
		return
	}
	success := !req.Reject
	digest := h.app.Crypto.OracleResponseDigest(req.DealID, req.Kadastr, success)
	sig, err := h.app.Crypto.SignGOST3410(h.app.OraclePriv, digest)
	if err != nil {
		writeErr(w, http.StatusInternalServerError,
			fmt.Errorf("sign oracle response: %w", err))
		return
	}
	cbErr := h.app.Engine.ProcessRosreestrCallback(
		req.DealID, req.Kadastr, success, sig,
	)
	expired := false
	if cbErr != nil {
		if errors.Is(cbErr, engine.ErrContractExpired) {
			expired = true
		} else {
			writeErr(w, mapEngineErr(cbErr), cbErr)
			return
		}
	}
	final, err := h.app.Engine.GetEscrow(req.DealID)
	if err != nil {
		writeErr(w, mapEngineErr(err), err)
		return
	}
	writeJSON(w, http.StatusOK, demoRunResponse{
		DealID:          req.DealID,
		Token:           *token,
		OracleSignature: base64.StdEncoding.EncodeToString(sig),
		ContractFinal:   final,
		Expired:         expired,
	})
}
type securityStatusResponse struct {
	NodeID           string `json:"node_id"`
	DemoMode         bool   `json:"demo_mode"`
	TotalContracts   int    `json:"total_contracts"`
	Released         int    `json:"released"`
	Refunded         int    `json:"refunded"`
	Locked           int    `json:"locked"`
	MerkleRoot       string `json:"merkle_root"`
	MerkleLeaves     int    `json:"merkle_leaves"`
	OracleConfigured bool   `json:"oracle_configured"`
	OracleTLS        bool   `json:"oracle_tls"`
	Timestamp        int64  `json:"timestamp"`
}
func (h *handlers) securityStatus(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodGet {
		writeErr(w, http.StatusMethodNotAllowed, errors.New("method not allowed"))
		return
	}
	contracts := make(map[string]*engine.EscrowContract)
	err := h.app.Ledger.IterateEscrows(
		func(dealID string, c *engine.EscrowContract) error {
			contracts[dealID] = c
			return nil
		},
	)
	if err != nil {
		writeErr(w, http.StatusInternalServerError, err)
		return
	}
	stats := securityStatusResponse{
		NodeID:           h.app.Config.NodeID,
		DemoMode:         h.app.Config.DemoMode,
		TotalContracts:   len(contracts),
		OracleConfigured: h.app.Config.OracleURL != "",
		OracleTLS:        h.app.Config.OracleTLS.Enabled,
		Timestamp:        time.Now().Unix(),
	}
	for _, c := range contracts {
		switch c.Status {
		case engine.StatusReleased:
			stats.Released++
		case engine.StatusRefunded:
			stats.Refunded++
		case engine.StatusLocked:
			stats.Locked++
		}
	}
	if len(contracts) > 0 {
		tree, err := audit.BuildTree(h.app.Crypto, contracts)
		if err != nil {
			writeErr(w, http.StatusInternalServerError,
				fmt.Errorf("build merkle: %w", err))
			return
		}
		stats.MerkleRoot = hex.EncodeToString(tree.Tree.RootHash())
		stats.MerkleLeaves = tree.Tree.Size()
	}
	writeJSON(w, http.StatusOK, stats)
}

func loadConfig(path string) (*Config, error) {

		cfg.ListenAddr = ":8081"

type Server struct {
	g         *crypto.GOSTProvider
	priv      *gost3410.PrivateKey
	pub       *gost3410.PublicKey
	logReqs   bool
	alwaysRej bool
	requestsTotal atomic.Uint64
	requestsOK    atomic.Uint64
	requestsRej   atomic.Uint64
}
func (s *Server) handleRegister(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodPost {
		http.Error(w, "method not allowed", http.StatusMethodNotAllowed)
		return
	}
	s.requestsTotal.Add(1)
	var req oracle.RegisterRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		http.Error(w, "bad request: "+err.Error(), http.StatusBadRequest)
		return
	}
	if req.DealID == "" || req.Kadastr == "" {
		http.Error(w, "deal_id and kadastr are required", http.StatusBadRequest)
		return
	}
	success := !s.alwaysRej
	digest := s.g.OracleResponseDigest(req.DealID, req.Kadastr, success)
	sig, err := s.g.SignGOST3410(s.priv, digest)
	if err != nil {
		http.Error(w, "sign: "+err.Error(), http.StatusInternalServerError)
		return
	}
	resp := oracle.RegisterResponse{
		DealID:    req.DealID,
		Kadastr:   req.Kadastr,
		Success:   success,
		Signature: sig,
	}
	w.Header().Set("Content-Type", "application/json")
	if err := json.NewEncoder(w).Encode(resp); err != nil {
		log.Printf("[oracle] encode response: %v", err)
	}
	if success {
		s.requestsOK.Add(1)
	} else {
		s.requestsRej.Add(1)
	}
	if s.logReqs {
		log.Printf("[oracle] deal=%s kadastr=%s success=%v",
			req.DealID, req.Kadastr, success)
	}
}
func (s *Server) handleHealthz(w http.ResponseWriter, r *http.Request) {

	fmt.Fprintf(w, "ok total=%d ok=%d rejected=%d\n",
		s.requestsTotal.Load(), s.requestsOK.Load(), s.requestsRej.Load())
}
func resolveKeypair(
	g *crypto.GOSTProvider,
	privHex string,
	generate bool,
) (*gost3410.PrivateKey, *gost3410.PublicKey, error) {
	if privHex == "" {
		if !generate {
			return nil, nil, errors.New(
				"private_key_hex is empty and generate_if_missing is false")
		}
		priv, pub, err := g.GenerateKeyPair()
		if err != nil {
			return nil, nil, err
		}
		log.Printf("[oracle] generated private: %s", hex.EncodeToString(priv.Raw()))
		log.Printf("[oracle] generated public : %s", hex.EncodeToString(pub.Raw()))
		return priv, pub, nil
	}
	raw, err := hex.DecodeString(privHex)
	if err != nil {
		return nil, nil, fmt.Errorf("decode private_key_hex: %w", err)
	}
	priv, err := g.NewPrivateKeyFromRaw(raw)
	if err != nil {
		return nil, nil, fmt.Errorf("parse private key: %w", err)
	}
	pub, err := priv.PublicKey()
	if err != nil {
		return nil, nil, fmt.Errorf("derive public key: %w", err)
	}
	log.Printf("[oracle] public key: %s", hex.EncodeToString(pub.Raw()))
	return priv, pub, nil
}
func buildTLSConfig(cfg TLSConf) (*tls.Config, error) {
	if _, err := os.Stat(cfg.CACertPath); os.IsNotExist(err) {
		if !cfg.GenerateIfMiss {
			return nil, fmt.Errorf("ca cert not found at %s", cfg.CACertPath)
		}
		log.Printf("[oracle] generating self-signed cert bundle in %s", cfg.CertDir)
		bundle, err := oracle.GenerateCertBundle(cfg.CertDir)
		if err != nil {
			return nil, fmt.Errorf("generate certs: %w", err)
		}
		cfg.CACertPath = bundle.CAPath
		cfg.ServerCertPath = bundle.ServerCertPath
		cfg.ServerKeyPath = bundle.ServerKeyPath
		log.Printf("[oracle] certs generated:\n  CA:     %s\n  server: %s\n  key:    %s",
			bundle.CAPath, bundle.ServerCertPath, bundle.ServerKeyPath)
		log.Printf("[oracle] client cert (for nis-node): %s / %s",
			bundle.ClientCertPath, bundle.ClientKeyPath)
	}
	caPEM, err := os.ReadFile(cfg.CACertPath)
	if err != nil {
		return nil, fmt.Errorf("read ca: %w", err)
	}
	caPool := x509.NewCertPool()
	if !caPool.AppendCertsFromPEM(caPEM) {
		return nil, errors.New("ca.crt has no valid certificates")
	}
	serverCert, err := tls.LoadX509KeyPair(cfg.ServerCertPath, cfg.ServerKeyPath)
	if err != nil {
		return nil, fmt.Errorf("load server pair: %w", err)
	}
	return &tls.Config{
		MinVersion:   tls.VersionTLS13,
		Certificates: []tls.Certificate{serverCert},
		ClientCAs:    caPool,
		ClientAuth:   tls.RequireAndVerifyClientCert,
	}, nil
}

	OraclePrivateHex string `yaml:"oracle_private_hex"`

func LoadConfig(path string) (*Config, error) {
	raw, err := os.ReadFile(path)
	if err != nil {
		return nil, fmt.Errorf("app: read config %q: %w", path, err)
	}
	var cfg Config
	dec := yaml.NewDecoder(bytes.NewReader(raw))
	dec.KnownFields(true)
	if err := dec.Decode(&cfg); err != nil {
		return nil, fmt.Errorf("app: parse config %q: %w", path, err)
	}
	if err := cfg.Validate(); err != nil {
		return nil, err
	}
	return &cfg, nil
}
func (c *Config) Validate() error {
	if c == nil {
		return fmt.Errorf("app: nil config")
	}
	if c.NodeID == "" {
		return fmt.Errorf("app: node_id is required")
	}
	if c.DBPath == "" {
		return fmt.Errorf("app: db_path is required")
	}
	if c.APIAddr == "" {
		c.APIAddr = ":8080"
	}
	if !c.DemoMode && c.Keys.GenerateIfMissing {
		return fmt.Errorf("app: keys.generate_if_missing allowed only in DemoMode")
	}
	if c.Keys.IssuerPrivateHex == "" &&
		c.Keys.IssuerPublicHex == "" &&
		!c.Keys.GenerateIfMissing {
		return fmt.Errorf(
			"app: provide issuer_private_hex, issuer_public_hex or generate_if_missing")
	}
	if c.Keys.OraclePrivateHex == "" &&
		c.Keys.OraclePublicHex == "" &&
		!c.Keys.GenerateIfMissing {
		return fmt.Errorf(
			"app: provide oracle_private_hex, oracle_public_hex or generate_if_missing")
	}
	return nil
}
type App struct {
	Config *Config
	Crypto *crypto.GOSTProvider
	Ledger *ledger.BadgerLedger
	Engine *engine.SmartContractEngine
	Oracle *oracle.Client 
	IssuerPub *gost3410.PublicKey
	OraclePub *gost3410.PublicKey
	IssuerPriv *gost3410.PrivateKey
	OraclePriv *gost3410.PrivateKey
}
func New(cfg *Config) (*App, error) {
	if cfg == nil {
		return nil, fmt.Errorf("app: nil config")
	}
	if err := cfg.Validate(); err != nil {
		return nil, err
	}
	g := crypto.NewGOSTProvider()
	generate := cfg.Keys.GenerateIfMissing && cfg.DemoMode
	issuerPriv, issuerPub, err := resolveKeypair(
		g,
		cfg.Keys.IssuerPrivateHex,
		cfg.Keys.IssuerPublicHex,
		generate,
		"issuer",
	)
	if err != nil {
		return nil, fmt.Errorf("app: issuer keys: %w", err)
	}
	oraclePriv, oraclePub, err := resolveKeypair(
		g,
		cfg.Keys.OraclePrivateHex,
		cfg.Keys.OraclePublicHex,
		generate,
		"oracle",
	)
	if err != nil {
		return nil, fmt.Errorf("app: oracle keys: %w", err)
	}
	bl, err := ledger.NewBadgerLedger(cfg.DBPath)
	if err != nil {
		return nil, fmt.Errorf("app: open ledger: %w", err)
	}
	eng, err := engine.NewSmartContractEngine(bl, g, issuerPub, oraclePub)
	if err != nil {
		_ = bl.Close()
		return nil, fmt.Errorf("app: new engine: %w", err)
	}
	a := &App{
		Config:     cfg,
		Crypto:     g,
		Ledger:     bl,
		Engine:     eng,
		IssuerPub:  issuerPub,
		OraclePub:  oraclePub,
		IssuerPriv: issuerPriv,
		OraclePriv: oraclePriv,
	}
	if cfg.OracleURL != "" {
		var oc *oracle.Client
		var err error
		if cfg.OracleTLS.Enabled {
			if cfg.OracleTLS.CACertPath == "" ||
				cfg.OracleTLS.ClientCertPath == "" ||
				cfg.OracleTLS.ClientKeyPath == "" {
				_ = a.Close()
				return nil, fmt.Errorf(
					"app: oracle_tls.enabled requires ca_cert_path, client_cert_path and client_key_path")
			}
			oc, err = oracle.NewMTLSClient(
				cfg.OracleURL,
				cfg.OracleTLS.CACertPath,
				cfg.OracleTLS.ClientCertPath,
				cfg.OracleTLS.ClientKeyPath,
			)
		} else {
			oc, err = oracle.NewClient(cfg.OracleURL, nil)
		}
		if err != nil {
			_ = a.Close()
			return nil, fmt.Errorf("app: oracle client: %w", err)
		}
		a.Oracle = oc
	}
	return a, nil
}
func (a *App) Close() error {
	if a == nil || a.Ledger == nil {
		return nil
	}
	return a.Ledger.Close()
}
func resolveKeypair(
	g *crypto.GOSTProvider,
	privHex, pubHex string,
	generate bool,
	role string,
) (*gost3410.PrivateKey, *gost3410.PublicKey, error) {
	if privHex != "" {
		raw, err := hex.DecodeString(privHex)
		if err != nil {
			return nil, nil, fmt.Errorf("decode %s private hex: %w", role, err)
		}
		priv, err := g.NewPrivateKeyFromRaw(raw)
		if err != nil {
			return nil, nil, fmt.Errorf("parse %s private: %w", role, err)
		}
		pub, err := priv.PublicKey()
		if err != nil {
			return nil, nil, fmt.Errorf("derive %s public: %w", role, err)
		}
		return priv, pub, nil
	}
	if pubHex != "" {
		raw, err := hex.DecodeString(pubHex)
		if err != nil {
			return nil, nil, fmt.Errorf("decode %s public hex: %w", role, err)
		}
		pub, err := g.NewPublicKeyFromRaw(raw)
		if err != nil {
			return nil, nil, fmt.Errorf("parse %s public: %w", role, err)
		}
		return nil, pub, nil
	}
	if generate {
		priv, pub, err := g.GenerateKeyPair()
		if err != nil {
			return nil, nil, err
		}
		fmt.Fprintf(os.Stderr, "[app] generated %s private: %s\n",
			role, hex.EncodeToString(priv.Raw()))
		fmt.Fprintf(os.Stderr, "[app] generated %s public : %s\n",
			role, hex.EncodeToString(pub.Raw()))
		return priv, pub, nil
	}
	return nil, nil, fmt.Errorf("no key material for %s", role)
}

import (
	"encoding/json"
	"sort"
	"nis-core/internal/crypto"

func LeafData(dealID string, c *engine.EscrowContract) []byte {
	rec := map[string]any{
		"deal_id":    dealID,
		"token_id":   c.Token.TokenID,
		"amount":     c.Token.Amount,
		"color_tag":  c.Token.TargetColorTag,
		"status":     string(c.Status),
		"created_at": c.CreatedAt,
		"updated_at": c.UpdatedAt,
		"timeout_at": c.TimeoutAt,
	}
	buf, _ := json.Marshal(rec)
	return buf
}
type Tree struct {
	Tree    *crypto.MerkleTree
	DealIDs []string 
}
func BuildTree(
	g *crypto.GOSTProvider,
	contracts map[string]*engine.EscrowContract,
) (*Tree, error) {
	keys := make([]string, 0, len(contracts))
	for k := range contracts {
		keys = append(keys, k)
	}
	sort.Strings(keys)
	leaves := make([][]byte, len(keys))
	for i, k := range keys {
		leaves[i] = LeafData(k, contracts[k])
	}
	t, err := crypto.BuildMerkleTree(g, leaves)
	if err != nil {
		return nil, err
	}
	return &Tree{Tree: t, DealIDs: keys}, nil
}
func (t *Tree) IndexOf(dealID string) int {
	for i, k := range t.DealIDs {
		if k == dealID {
			return i
		}
	}
	return -1
}

package crypto
import (
	"crypto"
	"crypto/cipher"
	"crypto/rand"
	"errors"
	"fmt"
	"hash"
	"io"
	"sync"
	"github.com/bi-zone/ruwireguard-go/crypto/gost/gost3410"
	"github.com/bi-zone/ruwireguard-go/crypto/gost/gost34112012256"
	"github.com/bi-zone/ruwireguard-go/crypto/gost/gost34112012512"
	"github.com/bi-zone/ruwireguard-go/crypto/gost/gost3412128"
	"github.com/bi-zone/ruwireguard-go/crypto/mgm"
)
const (
	KuznechikKeySize       = 32
	KuznechikBlockSize     = 16
	StreebogSize           = 32
	Streebog512Size        = 64
	MGMTagSize             = 16
	MGMNonceSize           = 16
	GOST3410SignatureSize  = 64
	GOST3410PrivateKeySize = 32
)
type GOSTProvider struct {
	curve *gost3410.Curve
	ecMu sync.Mutex
}
func NewGOSTProvider() *GOSTProvider {
	return &GOSTProvider{
		curve: gost3410.CurveIdtc26gost34102012256paramSetA(),
	}
}
func NewGOSTProviderWithCurve(curve *gost3410.Curve) *GOSTProvider {
	return &GOSTProvider{curve: curve}
}
func (g *GOSTProvider) Curve() *gost3410.Curve {
	return g.curve
}
func (g *GOSTProvider) Streebog256(data []byte) []byte {
	h := gost34112012256.New()
	h.Write(data)
	return h.Sum(nil)
}
func (g *GOSTProvider) NewStreebog256() hash.Hash {
	return gost34112012256.New()
}
func (g *GOSTProvider) Streebog512(data []byte) []byte {
	h := gost34112012512.New()
	h.Write(data)
	return h.Sum(nil)
}
func (g *GOSTProvider) NewStreebog512() hash.Hash {
	return gost34112012512.New()
}
func (g *GOSTProvider) GenerateKeyPair() (*gost3410.PrivateKey, *gost3410.PublicKey, error) {
	g.ecMu.Lock()
	defer g.ecMu.Unlock()
	priv, err := gost3410.GenPrivateKey(g.curve, rand.Reader)
	if err != nil {
		return nil, nil, fmt.Errorf("crypto: gen private key: %w", err)
	}
	pub, err := priv.PublicKey()
	if err != nil {
		return nil, nil, fmt.Errorf("crypto: derive public key: %w", err)
	}
	return priv, pub, nil
}
func (g *GOSTProvider) NewPrivateKeyFromRaw(raw []byte) (*gost3410.PrivateKey, error) {
	if len(raw) != GOST3410PrivateKeySize {
		return nil, fmt.Errorf("crypto: private key must be %d bytes, got %d",
			GOST3410PrivateKeySize, len(raw))
	}
	g.ecMu.Lock()
	defer g.ecMu.Unlock()
	return gost3410.NewPrivateKey(g.curve, raw)
}
func (g *GOSTProvider) NewPublicKeyFromRaw(raw []byte) (*gost3410.PublicKey, error) {
	g.ecMu.Lock()
	defer g.ecMu.Unlock()
	return gost3410.NewPublicKey(g.curve, raw)
}
func (g *GOSTProvider) SignGOST3410(
	priv *gost3410.PrivateKey,
	digest []byte,
) ([]byte, error) {
	if priv == nil {
		return nil, errors.New("crypto: private key is nil")
	}
	if len(digest) != StreebogSize {
		return nil, fmt.Errorf("crypto: digest must be %d bytes, got %d",
			StreebogSize, len(digest))
	}
	g.ecMu.Lock()
	defer g.ecMu.Unlock()
	sig, err := priv.SignDigest(digest, rand.Reader)
	if err != nil {
		return nil, fmt.Errorf("crypto: sign: %w", err)
	}
	if len(sig) != GOST3410SignatureSize {
		return nil, fmt.Errorf("crypto: unexpected signature size %d, want %d",
			len(sig), GOST3410SignatureSize)
	}
	return sig, nil
}
func (g *GOSTProvider) VerifyGOST3410(
	pub *gost3410.PublicKey,
	digest []byte,
	sig []byte,
) (bool, error) {
	if pub == nil {
		return false, errors.New("crypto: public key is nil")
	}
	if len(digest) != StreebogSize {
		return false, fmt.Errorf("crypto: digest must be %d bytes, got %d",
			StreebogSize, len(digest))
	}
	if len(sig) != GOST3410SignatureSize {
		return false, fmt.Errorf("crypto: signature must be %d bytes, got %d",
			GOST3410SignatureSize, len(sig))
	}
	g.ecMu.Lock()
	defer g.ecMu.Unlock()
	ok, err := pub.VerifyDigest(digest, sig)
	if err != nil {
		return false, fmt.Errorf("crypto: verify: %w", err)
	}
	return ok, nil
}
func (g *GOSTProvider) SignHash(
	priv *gost3410.PrivateKey,
	digest []byte,
	_ crypto.SignerOpts,
) ([]byte, error) {
	if priv == nil {
		return nil, errors.New("crypto: private key is nil")
	}
	g.ecMu.Lock()
	defer g.ecMu.Unlock()
	return priv.Sign(rand.Reader, digest, nil)
}
func newMGM(key []byte) (cipher.AEAD, error) {
	if len(key) != KuznechikKeySize {
		return nil, fmt.Errorf("crypto: kuznechik key must be %d bytes, got %d",
			KuznechikKeySize, len(key))
	}
	block := gost3412128.NewCipher(key)
	aead, err := mgm.NewMGM(block)
	if err != nil {
		return nil, fmt.Errorf("crypto: mgm init: %w", err)
	}
	return aead, nil
}
func validateMGMNonce(nonce []byte, wantSize int) error {
	if len(nonce) != wantSize {
		return fmt.Errorf("crypto: nonce must be %d bytes, got %d",
			wantSize, len(nonce))
	}
	if nonce[0]&0x80 != 0 {
		return errors.New("crypto: nonce top bit must be 0 (RFC 9058 MGM)")
	}
	return nil
}
func (g *GOSTProvider) SealMGM(
	key []byte,
	nonce []byte,
	plaintext []byte,
	additionalData []byte,
) ([]byte, error) {
	aead, err := newMGM(key)
	if err != nil {
		return nil, err
	}
	if err := validateMGMNonce(nonce, aead.NonceSize()); err != nil {
		return nil, err
	}
	if len(plaintext) == 0 && len(additionalData) == 0 {
		return nil, errors.New("crypto: MGM requires at least plaintext or additionalData")
	}
	return aead.Seal(nil, nonce, plaintext, additionalData), nil
}
func (g *GOSTProvider) OpenMGM(
	key []byte,
	nonce []byte,
	ciphertext []byte,
	additionalData []byte,
) ([]byte, error) {
	aead, err := newMGM(key)
	if err != nil {
		return nil, err
	}
	if err := validateMGMNonce(nonce, aead.NonceSize()); err != nil {
		return nil, err
	}
	if len(ciphertext) < aead.Overhead() {
		return nil, fmt.Errorf("crypto: ciphertext shorter than tag (%d < %d)",
			len(ciphertext), aead.Overhead())
	}
	pt, err := aead.Open(nil, nonce, ciphertext, additionalData)
	if err != nil {
		return nil, fmt.Errorf("crypto: mgm open: %w", err)
	}
	return pt, nil
}
func RandomBytes(n int) ([]byte, error) {
	b := make([]byte, n)
	if _, err := io.ReadFull(rand.Reader, b); err != nil {
		return nil, err
	}
	return b, nil
}
func RandomMGMNonce() ([]byte, error) {
	nonce, err := RandomBytes(MGMNonceSize)
	if err != nil {
		return nil, err
	}
	nonce[0] &= 0x7F
	return nonce, nil
}
func (g *GOSTProvider) OracleResponseDigest(dealID, kadastr string, success bool) []byte {
	return g.Streebog256(fmt.Appendf(nil, "%s:%s:%t", dealID, kadastr, success))
}

package crypto
import (
	"errors"
	"fmt"
)
type MerkleNode struct {
	Left     *MerkleNode
	Right    *MerkleNode
	Parent   *MerkleNode
	Hash     []byte
	LeafData []byte
}
type MerkleTree struct {
	Root   *MerkleNode
	Leaves []*MerkleNode
}
const (
	merkleLeafPrefix = 0x00
	merkleNodePrefix = 0x01
)
var ErrEmptyTree = errors.New("merkle: cannot build tree from empty slice")
func BuildMerkleTree(g *GOSTProvider, data [][]byte) (*MerkleTree, error) {
	if len(data) == 0 {
		return nil, ErrEmptyTree
	}
	leaves := make([]*MerkleNode, len(data))
	for i, d := range data {
		leaves[i] = &MerkleNode{
			Hash:     hashLeaf(g, d),
			LeafData: d,
		}
	}
	level := leaves
	for len(level) > 1 {
		if len(level)%2 == 1 {
			level = append(level, level[len(level)-1])
		}
		next := make([]*MerkleNode, 0, len(level)/2)
		for i := 0; i < len(level); i += 2 {
			parent := hashPair(g, level[i], level[i+1])
			level[i].Parent = parent
			level[i+1].Parent = parent
			next = append(next, parent)
		}
		level = next
	}
	return &MerkleTree{Root: level[0], Leaves: leaves}, nil
}
func hashLeaf(g *GOSTProvider, data []byte) []byte {
	buf := make([]byte, 0, 1+len(data))
	buf = append(buf, merkleLeafPrefix)
	buf = append(buf, data...)
	return g.Streebog256(buf)
}
func hashPair(g *GOSTProvider, left, right *MerkleNode) *MerkleNode {
	buf := make([]byte, 0, 1+len(left.Hash)+len(right.Hash))
	buf = append(buf, merkleNodePrefix)
	buf = append(buf, left.Hash...)
	buf = append(buf, right.Hash...)
	return &MerkleNode{Left: left, Right: right, Hash: g.Streebog256(buf)}
}
func (t *MerkleTree) RootHash() []byte {
	if t == nil || t.Root == nil {
		return nil
	}
	return t.Root.Hash
}
func (t *MerkleTree) Size() int {
	if t == nil {
		return 0
	}
	return len(t.Leaves)
}
type Proof struct {
	LeafHash []byte
	Siblings []ProofStep
}
type ProofStep struct {
	Hash   []byte
	IsLeft bool
}
func (t *MerkleTree) Prove(idx int) (*Proof, error) {
	if t == nil || t.Root == nil {
		return nil, errors.New("merkle: nil tree")
	}
	if idx < 0 || idx >= len(t.Leaves) {
		return nil, fmt.Errorf("merkle: leaf index %d out of range [0,%d)",
			idx, len(t.Leaves))
	}
	leaf := t.Leaves[idx]
	proof := &Proof{LeafHash: leaf.Hash}
	cur := leaf
	for cur.Parent != nil {
		parent := cur.Parent
		var sib *MerkleNode
		var isLeft bool
		if parent.Left == cur {
			sib = parent.Right
			isLeft = false
		} else {
			sib = parent.Left
			isLeft = true
		}
		proof.Siblings = append(proof.Siblings, ProofStep{Hash: sib.Hash, IsLeft: isLeft})
		cur = parent
	}
	return proof, nil
}
func VerifyProof(g *GOSTProvider, root []byte, proof *Proof) bool {
	if proof == nil || len(proof.LeafHash) == 0 {
		return false
	}
	cur := proof.LeafHash
	for _, step := range proof.Siblings {
		if step.IsLeft {
			cur = hashPair(g,
				&MerkleNode{Hash: step.Hash},
				&MerkleNode{Hash: cur},
			).Hash
		} else {
			cur = hashPair(g,
				&MerkleNode{Hash: cur},
				&MerkleNode{Hash: step.Hash},
			).Hash
		}
	}
	return bytesEqual(cur, root)
}
func bytesEqual(a, b []byte) bool {
	if len(a) != len(b) {
		return false
	}
	for i := range a {
		if a[i] != b[i] {
			return false
		}
	}
	return true
}

package engine
import (
	"crypto/subtle"
	"encoding/hex"
	"errors"
	"fmt"
	"time"
	"github.com/bi-zone/ruwireguard-go/crypto/gost/gost3410"
	"nis-core/internal/crypto"
)
const DefaultLockDuration = 30 * 24 * time.Hour
var (
	ErrTokenMisuse      = errors.New("engine: token color tag mismatch")
	ErrIssuerSigInvalid = errors.New("engine: issuer signature invalid")
	ErrOracleSigInvalid = errors.New("engine: oracle signature invalid")
	ErrContractExpired  = errors.New("engine: escrow contract expired")
	ErrInvalidState     = errors.New("engine: invalid state transition")
	ErrInvalidArgument  = errors.New("engine: invalid argument")
)
type SmartContractEngine struct {
	ledger  Ledger
	crypto  *crypto.GOSTProvider
	pubVoen *gost3410.PublicKey
	pubRosr *gost3410.PublicKey
	now     func() int64
}
func NewSmartContractEngine(
	ledger Ledger,
	g *crypto.GOSTProvider,
	pubVoen *gost3410.PublicKey,
	pubRosr *gost3410.PublicKey,
) (*SmartContractEngine, error) {
	if ledger == nil {
		return nil, fmt.Errorf("%w: nil ledger", ErrInvalidArgument)
	}
	if g == nil {
		return nil, fmt.Errorf("%w: nil crypto provider", ErrInvalidArgument)
	}
	if pubVoen == nil {
		return nil, fmt.Errorf("%w: nil issuer public key", ErrInvalidArgument)
	}
	if pubRosr == nil {
		return nil, fmt.Errorf("%w: nil oracle public key", ErrInvalidArgument)
	}
	return &SmartContractEngine{
		ledger:  ledger,
		crypto:  g,
		pubVoen: pubVoen,
		pubRosr: pubRosr,
		now:     func() int64 { return time.Now().Unix() },
	}, nil
}
func (e *SmartContractEngine) SetClock(fn func() int64) {
	if fn != nil {
		e.now = fn
	}
}
func (e *SmartContractEngine) Close() error {
	return e.ledger.Close()
}
func ComputeColorTag(g *crypto.GOSTProvider, inn, kadastr string) string {
	payload := inn + ":" + kadastr
	return hex.EncodeToString(g.Streebog256([]byte(payload)))
}
func (e *SmartContractEngine) issuerDigest(tokenID, colorTag string) []byte {
	return e.crypto.Streebog256([]byte(tokenID + ":" + colorTag))
}
func (e *SmartContractEngine) oracleDigest(dealID, kadastr string, success bool) []byte {
	return e.crypto.OracleResponseDigest(dealID, kadastr, success)
}
func newRandomID() (string, error) {
	b, err := crypto.RandomBytes(32)
	if err != nil {
		return "", fmt.Errorf("engine: random id: %w", err)
	}
	return hex.EncodeToString(b), nil
}
func (e *SmartContractEngine) IssueColoredRubles(
	amount uint64,
	militaryHash string,
	developerINN string,
	kadastr string,
	issuerPriv *gost3410.PrivateKey,
) (*DigitalNISRubles, error) {
	if amount == 0 {
		return nil, fmt.Errorf("%w: amount must be > 0", ErrInvalidArgument)
	}
	if militaryHash == "" {
		return nil, fmt.Errorf("%w: empty militaryHash", ErrInvalidArgument)
	}
	if developerINN == "" {
		return nil, fmt.Errorf("%w: empty developerINN", ErrInvalidArgument)
	}
	if kadastr == "" {
		return nil, fmt.Errorf("%w: empty kadastr", ErrInvalidArgument)
	}
	if issuerPriv == nil {
		return nil, fmt.Errorf("%w: nil issuer private key", ErrInvalidArgument)
	}
	colorTag := ComputeColorTag(e.crypto, developerINN, kadastr)
	nonce, err := newRandomID()
	if err != nil {
		return nil, err
	}
	tokenID := hex.EncodeToString(
		e.crypto.Streebog256([]byte(militaryHash + ":" + colorTag + ":" + nonce)),
	)
	token := &DigitalNISRubles{
		TokenID:         tokenID,
		Amount:          amount,
		TargetColorTag:  colorTag,
		MilitaryIDHash:  militaryHash,
		BeneficiaryINN:  developerINN,
		CadastralNumber: kadastr,
		IssuedAt:        e.now(),
	}
	digest := e.issuerDigest(token.TokenID, token.TargetColorTag)
	sig, err := e.crypto.SignGOST3410(issuerPriv, digest)
	if err != nil {
		return nil, fmt.Errorf("engine: sign issue: %w", err)
	}
	token.IssuerSignature = sig
	return token, nil
}
func (e *SmartContractEngine) LockInEscrow(
	dealID string,
	token *DigitalNISRubles,
	lockDuration time.Duration,
) error {
	if dealID == "" {
		return fmt.Errorf("%w: empty dealID", ErrInvalidArgument)
	}
	if token == nil {
		return fmt.Errorf("%w: nil token", ErrInvalidArgument)
	}
	if lockDuration <= 0 {
		return fmt.Errorf("%w: lockDuration must be > 0", ErrInvalidArgument)
	}
	if err := token.Validate(); err != nil {
		return fmt.Errorf("%w: token: %v", ErrInvalidArgument, err)
	}
	expectedTag := ComputeColorTag(e.crypto, token.BeneficiaryINN, token.CadastralNumber)
	if subtle.ConstantTimeCompare(
		[]byte(token.TargetColorTag),
		[]byte(expectedTag),
	) != 1 {
		return ErrTokenMisuse
	}
	digest := e.issuerDigest(token.TokenID, token.TargetColorTag)
	ok, err := e.crypto.VerifyGOST3410(e.pubVoen, digest, token.IssuerSignature)
	if err != nil {
		return fmt.Errorf("engine: verify issuer sig: %w", err)
	}
	if !ok {
		return ErrIssuerSigInvalid
	}
	now := e.now()
	contract := &EscrowContract{
		DealID:    dealID,
		Token:     *token,
		Status:    StatusLocked,
		CreatedAt: now,
		UpdatedAt: now,
		TimeoutAt: now + int64(lockDuration.Seconds()),
	}
	if err := e.ledger.PutEscrow(dealID, contract); err != nil {
		return fmt.Errorf("engine: lock in escrow: %w", err)
	}
	return nil
}
func (e *SmartContractEngine) ProcessRosreestrCallback(
	dealID string,
	kadastr string,
	success bool,
	oracleSig []byte,
) error {
	if dealID == "" {
		return fmt.Errorf("%w: empty dealID", ErrInvalidArgument)
	}
	if len(oracleSig) == 0 {
		return fmt.Errorf("%w: empty oracle signature", ErrInvalidArgument)
	}
	contract, err := e.ledger.GetEscrow(dealID)
	if err != nil {
		return err
	}
	if contract.Status != StatusLocked {
		return ErrInvalidState
	}
	if e.now() > contract.TimeoutAt {
		if err := e.RefundExpired(dealID); err != nil {
			return fmt.Errorf("engine: auto-refund after timeout: %w", err)
		}
		return ErrContractExpired
	}
	return e.ledger.UpdateEscrow(dealID, func(c *EscrowContract) error {
		if c.Status != StatusLocked {
			return ErrInvalidState
		}
		digest := e.oracleDigest(dealID, kadastr, success)
		ok, err := e.crypto.VerifyGOST3410(e.pubRosr, digest, oracleSig)
		if err != nil {
			return fmt.Errorf("engine: verify oracle sig: %w", err)
		}
		if !ok {
			return ErrOracleSigInvalid
		}
		if subtle.ConstantTimeCompare(
			[]byte(kadastr),
			[]byte(c.Token.CadastralNumber),
		) != 1 {
			return ErrTokenMisuse
		}
		if !success {
			return e.refundLocked(c, "rosreestr rejected registration")
		}
		txID, err := newRandomID()
		if err != nil {
			return err
		}
		c.Status = StatusReleased
		c.RosreestrProof = append([]byte(nil), oracleSig...)
		c.SettlementTxID = txID
		c.UpdatedAt = e.now()
		return nil
	})
}
func (e *SmartContractEngine) refundLocked(c *EscrowContract, reason string) error {
	c.Status = StatusRefunded
	c.RefundReason = reason
	c.UpdatedAt = e.now()
	return nil
}
func (e *SmartContractEngine) RefundExpired(dealID string) error {
	if dealID == "" {
		return fmt.Errorf("%w: empty dealID", ErrInvalidArgument)
	}
	return e.ledger.UpdateEscrow(dealID, func(c *EscrowContract) error {
		if c.Status == StatusRefunded || c.Status == StatusReleased {
			return nil
		}
		if c.Status != StatusLocked {
			return ErrInvalidState
		}
		if e.now() <= c.TimeoutAt {
			return nil 
		}
		return e.refundLocked(c, "timelock expired (forced)")
	})
}
func (e *SmartContractEngine) GetEscrow(dealID string) (*EscrowContract, error) {
	return e.ledger.GetEscrow(dealID)

	contracts map[string]*EscrowContract

	return &MockLedger{contracts: make(map[string]*EscrowContract)}
}
func (m *MockLedger) PutEscrow(dealID string, contract *EscrowContract) error {
	if dealID == "" {
		return errors.New("mock ledger: empty dealID")
	}
	if contract == nil {
		return errors.New("mock ledger: nil contract")
	}
	m.mu.Lock()
	defer m.mu.Unlock()
	if _, exists := m.contracts[dealID]; exists {
		return ErrConflict
	}
	m.contracts[dealID] = contract.Clone()
	return nil
}
func (m *MockLedger) GetEscrow(dealID string) (*EscrowContract, error) {
	m.mu.Lock()
	defer m.mu.Unlock()
	c, ok := m.contracts[dealID]
	if !ok {
		return nil, ErrNotFound
	}
	return c.Clone(), nil
}
func (m *MockLedger) UpdateEscrow(dealID string, fn func(*EscrowContract) error) error {
	if fn == nil {
		return errors.New("mock ledger: nil update function")
	}
	m.mu.Lock()
	defer m.mu.Unlock()
	c, ok := m.contracts[dealID]
	if !ok {
		return ErrNotFound
	}
	work := c.Clone()
	if err := fn(work); err != nil {
		return err
	}
	m.contracts[dealID] = work
	return nil
}

	return nil
}

package engine
import (
	"encoding/json"
	"errors"
	"fmt"
)
type EscrowStatus string
const (
	StatusCreated EscrowStatus = "CREATED"
	StatusLocked EscrowStatus = "LOCKED_IN_ESCROW"
	StatusReleased EscrowStatus = "RELEASED_TO_SELLER"
	StatusRefunded EscrowStatus = "REFUNDED_TO_MINOBORONA"
)
func (s EscrowStatus) IsValid() bool {
	switch s {
	case StatusCreated, StatusLocked, StatusReleased, StatusRefunded:
		return true
	}
	return false
}
type DigitalNISRubles struct {
	TokenID string `json:"token_id"`
	Amount uint64 `json:"amount"`
	TargetColorTag string `json:"target_color_tag"`
	MilitaryIDHash string `json:"military_id_hash"`
	BeneficiaryINN string `json:"beneficiary_inn"`
	CadastralNumber string `json:"cadastral_num"`
	IssuerSignature []byte `json:"issuer_signature"`
	IssuedAt int64 `json:"issued_at"`
}
type EscrowContract struct {
	DealID string `json:"deal_id"`
	Token DigitalNISRubles `json:"token"`
	Status EscrowStatus `json:"status"`
	CreatedAt int64 `json:"created_at"`
	UpdatedAt int64 `json:"updated_at"`
	TimeoutAt int64 `json:"timeout_at"`
	RosreestrProof []byte `json:"rosreestr_proof,omitempty"`
	SettlementTxID string `json:"settlement_tx_id,omitempty"`
	RefundReason string `json:"refund_reason,omitempty"`
}
func (c *EscrowContract) Serialize() ([]byte, error) {
	if c == nil {
		return nil, errors.New("engine: cannot serialize nil contract")
	}
	return json.Marshal(c)
}
func DeserializeEscrow(data []byte) (*EscrowContract, error) {
	if len(data) == 0 {
		return nil, errors.New("engine: cannot deserialize empty data")
	}
	var c EscrowContract
	if err := json.Unmarshal(data, &c); err != nil {
		return nil, fmt.Errorf("engine: deserialize escrow: %w", err)
	}
	return &c, nil
}
func (c *EscrowContract) Validate() error {
	if c == nil {
		return errors.New("engine: nil contract")
	}
	if c.DealID == "" {
		return errors.New("engine: empty DealID")
	}
	if !c.Status.IsValid() {
		return fmt.Errorf("engine: invalid status %q", c.Status)
	}
	if c.CreatedAt <= 0 {
		return errors.New("engine: CreatedAt must be positive")
	}
	if c.TimeoutAt <= c.CreatedAt {
		return errors.New("engine: TimeoutAt must be after CreatedAt")
	}
	if err := c.Token.Validate(); err != nil {
		return fmt.Errorf("engine: invalid token: %w", err)
	}
	return nil
}
func (t *DigitalNISRubles) Validate() error {
	if t == nil {
		return errors.New("engine: nil token")
	}
	if t.TokenID == "" {
		return errors.New("engine: empty TokenID")
	}
	if t.Amount == 0 {
		return errors.New("engine: amount must be positive")
	}
	if t.TargetColorTag == "" {
		return errors.New("engine: empty TargetColorTag")
	}
	if t.BeneficiaryINN == "" {
		return errors.New("engine: empty BeneficiaryINN")
	}
	if t.CadastralNumber == "" {
		return errors.New("engine: empty CadastralNumber")
	}
	if len(t.IssuerSignature) == 0 {
		return errors.New("engine: missing IssuerSignature")
	}
	return nil
}
func (c *EscrowContract) Clone() *EscrowContract {
	if c == nil {
		return nil
	}
	cp := *c
	if c.Token.IssuerSignature != nil {
		cp.Token.IssuerSignature = append([]byte(nil), c.Token.IssuerSignature...)
	}
	if c.RosreestrProof != nil {
		cp.RosreestrProof = append([]byte(nil), c.RosreestrProof...)
	}
	return &cp
}

package ledger
import (
	"errors"
	"fmt"
	badger "github.com/dgraph-io/badger/v4"
	"nis-core/internal/engine"
)
const EscrowKeyPrefix = "escrow:"
type BadgerLedger struct {
	db *badger.DB
}
var _ engine.Ledger = (*BadgerLedger)(nil)
func NewBadgerLedger(path string) (*BadgerLedger, error) {
	if path == "" {
		return nil, errors.New("ledger: empty db path")
	}
	opts := badger.DefaultOptions(path).
		WithLogger(nil).
		WithLoggingLevel(badger.ERROR).
		WithValueLogFileSize(64 << 20)
	db, err := badger.Open(opts)
	if err != nil {
		return nil, fmt.Errorf("ledger: open badger at %q: %w", path, err)
	}
	return &BadgerLedger{db: db}, nil
}
func (l *BadgerLedger) Close() error {
	if l == nil || l.db == nil {
		return nil
	}
	return l.db.Close()
}
func escrowKey(dealID string) []byte {
	return []byte(EscrowKeyPrefix + dealID)
}
func (l *BadgerLedger) PutEscrow(dealID string, contract *engine.EscrowContract) error {
	if dealID == "" {
		return errors.New("ledger: empty dealID")
	}
	if contract == nil {
		return errors.New("ledger: nil contract")
	}
	payload, err := contract.Serialize()
	if err != nil {
		return fmt.Errorf("ledger: serialize contract: %w", err)
	}
	key := escrowKey(dealID)
	return l.db.Update(func(txn *badger.Txn) error {
		_, err := txn.Get(key)
		switch {
		case err == nil:
			return engine.ErrConflict
		case errors.Is(err, badger.ErrKeyNotFound):
		default:
			return fmt.Errorf("ledger: check existence: %w", err)
		}
		if err := txn.Set(key, payload); err != nil {
			return fmt.Errorf("ledger: set contract: %w", err)
		}
		return nil
	})
}
func (l *BadgerLedger) GetEscrow(dealID string) (*engine.EscrowContract, error) {
	if dealID == "" {
		return nil, errors.New("ledger: empty dealID")
	}
	key := escrowKey(dealID)
	var payload []byte
	err := l.db.View(func(txn *badger.Txn) error {
		item, err := txn.Get(key)
		if err != nil {
			return err
		}
		return item.Value(func(v []byte) error {
			payload = append([]byte(nil), v...)
			return nil
		})
	})
	if err != nil {
		if errors.Is(err, badger.ErrKeyNotFound) {
			return nil, engine.ErrNotFound
		}
		return nil, fmt.Errorf("ledger: get escrow: %w", err)
	}
	contract, err := engine.DeserializeEscrow(payload)
	if err != nil {
		return nil, fmt.Errorf("ledger: deserialize escrow: %w", err)
	}
	return contract, nil
}
func (l *BadgerLedger) UpdateEscrow(dealID string, fn func(*engine.EscrowContract) error) error {
	if dealID == "" {
		return errors.New("ledger: empty dealID")
	}
	if fn == nil {
		return errors.New("ledger: nil update function")
	}
	key := escrowKey(dealID)
	return l.db.Update(func(txn *badger.Txn) error {
		item, err := txn.Get(key)
		if err != nil {
			if errors.Is(err, badger.ErrKeyNotFound) {
				return engine.ErrNotFound
			}
			return fmt.Errorf("ledger: get for update: %w", err)
		}
		var payload []byte
		if err := item.Value(func(v []byte) error {
			payload = append([]byte(nil), v...)
			return nil
		}); err != nil {
			return fmt.Errorf("ledger: read value: %w", err)
		}
		contract, err := engine.DeserializeEscrow(payload)
		if err != nil {
			return fmt.Errorf("ledger: deserialize: %w", err)
		}
		if err := fn(contract); err != nil {
			return err
		}
		updated, err := contract.Serialize()
		if err != nil {
			return fmt.Errorf("ledger: serialize after update: %w", err)
		}
		if err := txn.Set(key, updated); err != nil {
			return fmt.Errorf("ledger: set after update: %w", err)
		}
		return nil
	})
}
func (l *BadgerLedger) IterateEscrows(
	fn func(dealID string, c *engine.EscrowContract) error,
) error {
	if fn == nil {
		return errors.New("ledger: nil iteration function")
	}
	return l.db.View(func(txn *badger.Txn) error {
		it := txn.NewIterator(badger.DefaultIteratorOptions)
		defer it.Close()
		prefix := []byte(EscrowKeyPrefix)
		for it.Seek(prefix); it.ValidForPrefix(prefix); it.Next() {
			item := it.Item()
			key := string(item.Key())
			dealID := key[len(EscrowKeyPrefix):]
			var payload []byte
			if err := item.Value(func(v []byte) error {
				payload = append([]byte(nil), v...)
				return nil
			}); err != nil {
				return fmt.Errorf("ledger: read value %s: %w", dealID, err)
			}
			c, err := engine.DeserializeEscrow(payload)
			if err != nil {
				return fmt.Errorf("ledger: deserialize %s: %w", dealID, err)
			}
			if err := fn(dealID, c); err != nil {
				return err
			}
		}
		return nil
	})
}

package oracle
import (
	"bytes"
	"context"
	"crypto/tls"
	"crypto/x509"
	"encoding/json"
	"errors"
	"fmt"
	"io"
	"net/http"
	"os"
	"time"
)
const (
	DefaultTimeout = 10 * time.Second
	maxResponseSize = 1 << 20
	pathRegister = "/api/v1/register"
)
var (
	ErrNilRequest     = errors.New("oracle: nil request")
	ErrEmptyBaseURL   = errors.New("oracle: empty base URL")
	ErrHTTPStatus     = errors.New("oracle: unexpected HTTP status")
	ErrEmptyResponse  = errors.New("oracle: empty response body")
	ErrResponseTooBig = errors.New("oracle: response body exceeds limit")
	ErrTLSCertLoad    = errors.New("oracle: cannot load TLS certificate")
)
type RegisterRequest struct {
	DealID  string `json:"deal_id"`
	Kadastr string `json:"kadastr"`
}
type RegisterResponse struct {
	DealID    string `json:"deal_id"`
	Kadastr   string `json:"kadastr"`
	Success   bool   `json:"success"`
	Signature []byte `json:"signature"` 
}
type Client struct {
	baseURL string
	http    *http.Client
}
func NewClient(baseURL string, httpClient *http.Client) (*Client, error) {
	if baseURL == "" {
		return nil, ErrEmptyBaseURL
	}
	if httpClient == nil {
		httpClient = &http.Client{Timeout: DefaultTimeout}
	}
	return &Client{baseURL: baseURL, http: httpClient}, nil
}
func NewMTLSClient(
	baseURL, caCertPath, clientCertPath, clientKeyPath string,
) (*Client, error) {
	if baseURL == "" {
		return nil, ErrEmptyBaseURL
	}
	caPEM, err := os.ReadFile(caCertPath)
	if err != nil {
		return nil, fmt.Errorf("%w: ca.crt: %v", ErrTLSCertLoad, err)
	}
	caPool := x509.NewCertPool()
	if !caPool.AppendCertsFromPEM(caPEM) {
		return nil, fmt.Errorf("%w: ca.crt has no valid certificates", ErrTLSCertLoad)
	}
	clientCert, err := tls.LoadX509KeyPair(clientCertPath, clientKeyPath)
	if err != nil {
		return nil, fmt.Errorf("%w: client pair: %v", ErrTLSCertLoad, err)
	}
	tlsCfg := &tls.Config{
		MinVersion:   tls.VersionTLS13,
		RootCAs:      caPool,
		Certificates: []tls.Certificate{clientCert},
		ServerName:   "localhost",
	}
	return &Client{
		baseURL: baseURL,
		http: &http.Client{
			Timeout: DefaultTimeout,
			Transport: &http.Transport{
				TLSClientConfig: tlsCfg,
			},
		},
	}, nil
}
func (c *Client) Register(ctx context.Context, req *RegisterRequest) (*RegisterResponse, error) {
	if req == nil {
		return nil, ErrNilRequest
	}
	if req.DealID == "" || req.Kadastr == "" {
		return nil, fmt.Errorf("oracle: dealID and kadastr are required")
	}
	body, err := json.Marshal(req)
	if err != nil {
		return nil, fmt.Errorf("oracle: marshal request: %w", err)
	}
	httpReq, err := http.NewRequestWithContext(
		ctx, http.MethodPost, c.baseURL+pathRegister, bytes.NewReader(body),
	)
	if err != nil {
		return nil, fmt.Errorf("oracle: build request: %w", err)
	}
	httpReq.Header.Set("Content-Type", "application/json")
	httpReq.Header.Set("Accept", "application/json")
	resp, err := c.http.Do(httpReq)
	if err != nil {
		return nil, fmt.Errorf("oracle: do request: %w", err)
	}
	defer resp.Body.Close()
	limited := io.LimitReader(resp.Body, maxResponseSize+1)
	respBody, err := io.ReadAll(limited)
	if err != nil {
		return nil, fmt.Errorf("oracle: read body: %w", err)
	}
	if int64(len(respBody)) > maxResponseSize {
		return nil, ErrResponseTooBig
	}
	if resp.StatusCode != http.StatusOK {
		return nil, fmt.Errorf("%w: %d %s",
			ErrHTTPStatus, resp.StatusCode, string(respBody))
	}
	if len(respBody) == 0 {
		return nil, ErrEmptyResponse
	}
	var out RegisterResponse
	if err := json.Unmarshal(respBody, &out); err != nil {
		return nil, fmt.Errorf("oracle: unmarshal response: %w", err)
	}
	return &out, nil
}

package oracle
import (
	"crypto/ecdsa"
	"crypto/elliptic"
	"crypto/rand"
	"crypto/x509"
	"crypto/x509/pkix"
	"encoding/pem"
	"errors"
	"fmt"
	"math/big"
	"net"
	"os"
	"path/filepath"
	"time"
)
const (
	CertValidity = 365 * 24 * time.Hour
	CertOrg = "Digital NIS Prototype"
)
func certCurve() elliptic.Curve {
	return elliptic.P256()
}
type CertBundle struct {
	CAPath         string
	CAKeyPath      string
	ServerCertPath string
	ServerKeyPath  string
	ClientCertPath string
	ClientKeyPath  string
}
func GenerateCertBundle(dir string) (*CertBundle, error) {
	if dir == "" {
		return nil, errors.New("oracle: empty cert dir")
	}
	if err := os.MkdirAll(dir, 0o755); err != nil {
		return nil, fmt.Errorf("oracle: create cert dir: %w", err)
	}
	bundle := &CertBundle{
		CAPath:         filepath.Join(dir, "ca.crt"),
		CAKeyPath:      filepath.Join(dir, "ca.key"),
		ServerCertPath: filepath.Join(dir, "server.crt"),
		ServerKeyPath:  filepath.Join(dir, "server.key"),
		ClientCertPath: filepath.Join(dir, "client.crt"),
		ClientKeyPath:  filepath.Join(dir, "client.key"),
	}
	caKey, err := ecdsa.GenerateKey(certCurve(), rand.Reader)
	if err != nil {
		return nil, fmt.Errorf("oracle: gen ca key: %w", err)
	}
	caTmpl := &x509.Certificate{
		SerialNumber:          serialNumber(),
		Subject:               pkix.Name{Organization: []string{CertOrg + " CA"}, CommonName: "NIS Root CA"},
		NotBefore:             time.Now().Add(-time.Hour),
		NotAfter:              time.Now().Add(CertValidity * 5),
		KeyUsage:              x509.KeyUsageCertSign | x509.KeyUsageDigitalSignature,
		BasicConstraintsValid: true,
		IsCA:                  true,
		MaxPathLen:            1,
	}
	caDER, err := x509.CreateCertificate(rand.Reader, caTmpl, caTmpl, &caKey.PublicKey, caKey)
	if err != nil {
		return nil, fmt.Errorf("oracle: create ca cert: %w", err)
	}
	caCert, err := x509.ParseCertificate(caDER)
	if err != nil {
		return nil, fmt.Errorf("oracle: parse ca cert: %w", err)
	}
	if err := writeCertAndKey(bundle.CAPath, bundle.CAKeyPath, caDER, caKey); err != nil {
		return nil, err
	}
	if err := issueLeaf(
		caCert, caKey,
		bundle.ServerCertPath, bundle.ServerKeyPath,
		"nis-oracle-server", true,
	); err != nil {
		return nil, err
	}
	if err := issueLeaf(
		caCert, caKey,
		bundle.ClientCertPath, bundle.ClientKeyPath,
		"nis-node-client", false,
	); err != nil {
		return nil, err
	}
	return bundle, nil
}
func issueLeaf(
	caCert *x509.Certificate,
	caKey *ecdsa.PrivateKey,
	certPath, keyPath, cn string,
	isServer bool,
) error {
	leafKey, err := ecdsa.GenerateKey(certCurve(), rand.Reader)
	if err != nil {
		return fmt.Errorf("oracle: gen leaf key: %w", err)
	}
	tmpl := &x509.Certificate{
		SerialNumber: serialNumber(),
		Subject:      pkix.Name{Organization: []string{CertOrg}, CommonName: cn},
		NotBefore:    time.Now().Add(-time.Hour),
		NotAfter:     time.Now().Add(CertValidity),
		KeyUsage:     x509.KeyUsageDigitalSignature | x509.KeyUsageKeyEncipherment,
	}
	if isServer {
		tmpl.ExtKeyUsage = []x509.ExtKeyUsage{x509.ExtKeyUsageServerAuth}
		tmpl.DNSNames = []string{"localhost"}
		tmpl.IPAddresses = []net.IP{net.ParseIP("127.0.0.1"), net.ParseIP("::1")}
	} else {
		tmpl.ExtKeyUsage = []x509.ExtKeyUsage{x509.ExtKeyUsageClientAuth}
	}
	der, err := x509.CreateCertificate(rand.Reader, tmpl, caCert, &leafKey.PublicKey, caKey)
	if err != nil {
		return fmt.Errorf("oracle: create leaf cert: %w", err)
	}
	return writeCertAndKey(certPath, keyPath, der, leafKey)
}
func writeCertAndKey(certPath, keyPath string, der []byte, key *ecdsa.PrivateKey) error {
	certPEM := pem.EncodeToMemory(&pem.Block{Type: "CERTIFICATE", Bytes: der})
	if err := os.WriteFile(certPath, certPEM, 0o644); err != nil {
		return fmt.Errorf("oracle: write cert %q: %w", certPath, err)
	}
	keyDER, err := x509.MarshalPKCS8PrivateKey(key)
	if err != nil {
		return fmt.Errorf("oracle: marshal key: %w", err)
	}
	keyPEM := pem.EncodeToMemory(&pem.Block{Type: "PRIVATE KEY", Bytes: keyDER})
	if err := os.WriteFile(keyPath, keyPEM, 0o600); err != nil {
		return fmt.Errorf("oracle: write key %q: %w", keyPath, err)
	}
	return nil
}
func serialNumber() *big.Int {
	max := new(big.Int).Lsh(big.NewInt(1), 128)
	n, _ := rand.Int(rand.Reader, max)
	return n
}

package oracle
import (
	"crypto/subtle"
	"errors"
	"fmt"
	"github.com/bi-zone/ruwireguard-go/crypto/gost/gost3410"
	"nis-core/internal/crypto"
)
var (
	ErrNilVerifier     = errors.New("oracle: nil verifier")
	ErrNilResponse     = errors.New("oracle: nil response")
	ErrDealMismatch    = errors.New("oracle: response dealID does not match request")
	ErrKadastrMismatch = errors.New("oracle: response kadastr does not match request")
	ErrBadSignature    = errors.New("oracle: signature verification failed")
	ErrEmptySignature  = errors.New("oracle: empty signature")
)
type Verifier struct {
	g      *crypto.GOSTProvider
	pubKey *gost3410.PublicKey
}
func NewVerifier(g *crypto.GOSTProvider, pub *gost3410.PublicKey) (*Verifier, error) {
	if g == nil {
		return nil, errors.New("oracle: nil crypto provider")
	}
	if pub == nil {
		return nil, errors.New("oracle: nil public key")
	}
	return &Verifier{g: g, pubKey: pub}, nil
}
func (v *Verifier) VerifyResponse(
	resp *RegisterResponse,
	expectedDealID, expectedKadastr string,
) error {
	if v == nil {
		return ErrNilVerifier
	}
	if resp == nil {
		return ErrNilResponse
	}
	if len(resp.Signature) == 0 {
		return ErrEmptySignature
	}
	if len(resp.Signature) != crypto.GOST3410SignatureSize {
		return fmt.Errorf("oracle: signature must be %d bytes, got %d",
			crypto.GOST3410SignatureSize, len(resp.Signature))
	}
	if subtle.ConstantTimeCompare(
		[]byte(resp.DealID), []byte(expectedDealID),
	) != 1 {
		return ErrDealMismatch
	}
	if subtle.ConstantTimeCompare(
		[]byte(resp.Kadastr), []byte(expectedKadastr),
	) != 1 {
		return ErrKadastrMismatch
	}
	digest := v.g.OracleResponseDigest(resp.DealID, resp.Kadastr, resp.Success)
	ok, err := v.g.VerifyGOST3410(v.pubKey, digest, resp.Signature)
	if err != nil {
		return fmt.Errorf("oracle: verify: %w", err)
	}
	if !ok {
		return ErrBadSignature
	}
	return nil
}

package security
import (
	"encoding/hex"
	"errors"
	"fmt"
	"github.com/bi-zone/ruwireguard-go/crypto/gost/gost3410"
	"nis-core/internal/crypto"
)
const (
	NonceSize = 16
	PayloadHashSize = crypto.StreebogSize
	EnvelopeSignatureSize = crypto.GOST3410SignatureSize
)
var (
	ErrEnvelopeNil            = errors.New("security: nil envelope")
	ErrEnvelopeEmptySender    = errors.New("security: empty sender node id")
	ErrEnvelopeBadNonce       = errors.New("security: nonce must be hex of 16 bytes")
	ErrEnvelopeBadPayloadHash = errors.New("security: payload hash must be hex of 32 bytes")
	ErrEnvelopeBadSignature   = errors.New("security: signature must be 64 bytes")
)
type ZeroTrustEnvelope struct {
	SenderNodeID string `json:"sender_node_id"`
	Timestamp    int64  `json:"timestamp"`
	Nonce        string `json:"nonce"`        
	PayloadHash  string `json:"payload_hash"` 
	Signature    []byte `json:"signature"`    
}
func EnvelopeDigest(
	g *crypto.GOSTProvider,
	senderID string,
	timestamp int64,
	nonceHex, payloadHashHex string,
) []byte {
	payload := fmt.Sprintf("%s:%d:%s:%s", senderID, timestamp, nonceHex, payloadHashHex)
	return g.Streebog256([]byte(payload))
}
func BuildEnvelope(
	g *crypto.GOSTProvider,
	senderID string,
	timestamp int64,
	senderPriv *gost3410.PrivateKey,
	payload []byte,
) (*ZeroTrustEnvelope, error) {
	if senderID == "" {
		return nil, ErrEnvelopeEmptySender
	}
	if g == nil {
		return nil, errors.New("security: nil crypto provider")
	}
	if senderPriv == nil {
		return nil, errors.New("security: nil sender private key")
	}
	nonceBytes, err := crypto.RandomBytes(NonceSize)
	if err != nil {
		return nil, fmt.Errorf("security: generate nonce: %w", err)
	}
	env := &ZeroTrustEnvelope{
		SenderNodeID: senderID,
		Timestamp:    timestamp,
		Nonce:        hex.EncodeToString(nonceBytes),
		PayloadHash:  hex.EncodeToString(g.Streebog256(payload)),
	}
	digest := EnvelopeDigest(g, env.SenderNodeID, env.Timestamp, env.Nonce, env.PayloadHash)
	sig, err := g.SignGOST3410(senderPriv, digest)
	if err != nil {
		return nil, fmt.Errorf("security: sign envelope: %w", err)
	}
	env.Signature = sig
	return env, nil
}
func (e *ZeroTrustEnvelope) Validate() error {
	if e == nil {
		return ErrEnvelopeNil
	}
	if e.SenderNodeID == "" {
		return ErrEnvelopeEmptySender
	}
	nonceBytes, err := hex.DecodeString(e.Nonce)
	if err != nil || len(nonceBytes) != NonceSize {
		return ErrEnvelopeBadNonce
	}
	hashBytes, err := hex.DecodeString(e.PayloadHash)
	if err != nil || len(hashBytes) != PayloadHashSize {
		return ErrEnvelopeBadPayloadHash
	}
	if len(e.Signature) != EnvelopeSignatureSize {
		return ErrEnvelopeBadSignature
	}
	return nil
}

package security
import (
	"crypto/subtle"
	"encoding/hex"
	"errors"
	"fmt"
	"sync"
	"time"
	"github.com/bi-zone/ruwireguard-go/crypto/gost/gost3410"
	"nis-core/internal/crypto"
)
const (
	DefaultWindowSec = 15
	defaultNonceTTL = 2 * DefaultWindowSec
	nonceCacheSoftLimit = 128
)
var (
	ErrUnknownNode  = errors.New("security: unknown node")
	ErrReplayWindow = errors.New("security: timestamp outside allowed window")
	ErrPayloadHash  = errors.New("security: payload hash mismatch")
	ErrSignature    = errors.New("security: signature verification failed")
	ErrReplayNonce  = errors.New("security: nonce already used (replay)")
)
type ZeroTrustGateway struct {
	crypto *crypto.GOSTProvider
	mu          sync.Mutex
	trustedKeys map[string]*gost3410.PublicKey
	nonceCache  map[string]int64 
	windowSec int64
	nonceTTL  int64
	now       func() int64
}
func NewZeroTrustGateway(g *crypto.GOSTProvider) *ZeroTrustGateway {
	return &ZeroTrustGateway{
		crypto:      g,
		trustedKeys: make(map[string]*gost3410.PublicKey),
		nonceCache:  make(map[string]int64),
		windowSec:   DefaultWindowSec,
		nonceTTL:    defaultNonceTTL,
		now:         func() int64 { return time.Now().Unix() },
	}
}
func (gt *ZeroTrustGateway) RegisterTrustedNode(nodeID string, pub *gost3410.PublicKey) {
	if nodeID == "" || pub == nil {
		return
	}
	gt.mu.Lock()
	defer gt.mu.Unlock()
	gt.trustedKeys[nodeID] = pub
}
func (gt *ZeroTrustGateway) UnregisterTrustedNode(nodeID string) {
	gt.mu.Lock()
	defer gt.mu.Unlock()
	delete(gt.trustedKeys, nodeID)
}
func (gt *ZeroTrustGateway) SetClock(fn func() int64) {
	if fn == nil {
		return
	}
	gt.mu.Lock()
	defer gt.mu.Unlock()
	gt.now = fn
}
func (gt *ZeroTrustGateway) AuthorizeRequest(env *ZeroTrustEnvelope, rawPayload []byte) error {
	if gt == nil {
		return errors.New("security: nil gateway")
	}
	if err := env.Validate(); err != nil {
		return err
	}
	gt.mu.Lock()
	pub, ok := gt.trustedKeys[env.SenderNodeID]
	now := gt.now()
	window := gt.windowSec
	nonceTTL := gt.nonceTTL
	gt.mu.Unlock()
	if !ok {
		return fmt.Errorf("%w: %s", ErrUnknownNode, env.SenderNodeID)
	}
	delta := now - env.Timestamp
	if delta < 0 {
		delta = -delta
	}
	if delta > window {
		return fmt.Errorf("%w: delta=%ds, window=%ds",
			ErrReplayWindow, delta, window)
	}
	expectedHash := hex.EncodeToString(gt.crypto.Streebog256(rawPayload))
	if subtle.ConstantTimeCompare([]byte(expectedHash), []byte(env.PayloadHash)) != 1 {
		return ErrPayloadHash
	}
	digest := EnvelopeDigest(gt.crypto, env.SenderNodeID, env.Timestamp, env.Nonce, env.PayloadHash)
	ok, err := gt.crypto.VerifyGOST3410(pub, digest, env.Signature)
	if err != nil {
		return fmt.Errorf("security: verify signature: %w", err)
	}
	if !ok {
		return ErrSignature
	}
	gt.mu.Lock()
	defer gt.mu.Unlock()
	gt.evictExpiredLocked(now)
	if _, seen := gt.nonceCache[env.Nonce]; seen {
		return fmt.Errorf("%w: %s", ErrReplayNonce, env.Nonce)
	}
	gt.nonceCache[env.Nonce] = now + nonceTTL
	return nil
}
func (gt *ZeroTrustGateway) evictExpiredLocked(now int64) {
	if len(gt.nonceCache) < nonceCacheSoftLimit {
		return
	}
	for k, exp := range gt.nonceCache {
		if exp <= now {
			delete(gt.nonceCache, k)
		}
	}
}
func (gt *ZeroTrustGateway) NonceCacheSize() int {
	gt.mu.Lock()
	defer gt.mu.Unlock()
	return len(gt.nonceCache)
}

		t.Fatalf("err = %v, want ErrReplayNonce", err)
	}
}
func TestZeroTrustGateway_NonceEviction(t *testing.T) {
	env := newTestEnv(t)
	env.gw.mu.Lock()
	for i := 0; i < 200; i++ {
		env.gw.nonceCache[fmt.Sprintf("stale-%d", i)] = env.clock.Now() - 1
	}
	env.gw.mu.Unlock()
	payload := []byte("x")
	envelope, _ := BuildEnvelope(env.g, "node-A", env.clock.Now(), env.priv, payload)
	if err := env.gw.AuthorizeRequest(envelope, payload); err != nil {
		t.Fatalf("authorize: %v", err)
	}
	if size := env.gw.NonceCacheSize(); size != 1 {
		t.Fatalf("cache size = %d, want 1 after eviction", size)
	}
}

func TestZeroTrustEnvelope_Validate_Structural(t *testing.T) {
	g := crypto.NewGOSTProvider()
	priv, _, _ := g.GenerateKeyPair()
	valid, err := BuildEnvelope(g, "node-A", 1_700_000_000, priv, []byte("x"))
	if err != nil {
		t.Fatalf("BuildEnvelope: %v", err)
	}
	t.Run("nil", func(t *testing.T) {
		var nilEnv *ZeroTrustEnvelope
		if err := nilEnv.Validate(); err == nil {
			t.Fatal("expected error for nil envelope")
		}
	})
	cases := []struct {
		name   string
		mutate func(*ZeroTrustEnvelope)
	}{
		{"empty sender", func(e *ZeroTrustEnvelope) { e.SenderNodeID = "" }},
		{"bad nonce hex", func(e *ZeroTrustEnvelope) { e.Nonce = "zz" }},
		{"short nonce", func(e *ZeroTrustEnvelope) { e.Nonce = "0011" }},
		{"bad payload hash", func(e *ZeroTrustEnvelope) { e.PayloadHash = "not-hex" }},
		{"short payload hash", func(e *ZeroTrustEnvelope) { e.PayloadHash = "0011" }},
		{"short signature", func(e *ZeroTrustEnvelope) { e.Signature = []byte{1, 2, 3} }},
	}
	for _, c := range cases {
		t.Run(c.name, func(t *testing.T) {
			cp := *valid
			cp.Signature = append([]byte(nil), valid.Signature...)
			c.mutate(&cp)
			if err := cp.Validate(); err == nil {
				t.Fatal("expected structural error")
			}
		})
	}
}
